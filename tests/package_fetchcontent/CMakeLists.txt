cmake_minimum_required(VERSION 3.21)
project(mwfl_fetchcontent_consumer LANGUAGES CXX)
include(FetchContent)
set(MWFL_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(MWFL_BUILD_TESTS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(mwfl SOURCE_DIR "${MWFL_SOURCE_DIR}")
FetchContent_MakeAvailable(mwfl)
add_executable(mwfl_fetchcontent_consumer ../package_consumer/main.cpp)
target_compile_features(mwfl_fetchcontent_consumer PRIVATE cxx_std_20)
target_link_libraries(mwfl_fetchcontent_consumer PRIVATE mwfl::ui)
add_executable(mwfl_fetchcontent_printing_consumer ../package_consumer/printing_main.cpp)
target_compile_features(mwfl_fetchcontent_printing_consumer PRIVATE cxx_std_20)
target_link_libraries(mwfl_fetchcontent_printing_consumer PRIVATE mwfl::printing)
add_executable(mwfl_fetchcontent_ole_consumer ../package_consumer/ole_main.cpp)
target_compile_features(mwfl_fetchcontent_ole_consumer PRIVATE cxx_std_20)
target_link_libraries(mwfl_fetchcontent_ole_consumer PRIVATE mwfl::ole)
add_executable(mwfl_fetchcontent_shell_consumer ../package_consumer/shell_main.cpp)
target_compile_features(mwfl_fetchcontent_shell_consumer PRIVATE cxx_std_20)
target_link_libraries(mwfl_fetchcontent_shell_consumer PRIVATE mwfl::shell)
foreach(component IN ITEMS ribbon mdi graphics)
    add_executable(mwfl_fetchcontent_${component}_consumer
        ../package_consumer/${component}_main.cpp)
    target_compile_features(mwfl_fetchcontent_${component}_consumer PRIVATE cxx_std_20)
    target_link_libraries(mwfl_fetchcontent_${component}_consumer PRIVATE mwfl::${component})
endforeach()
