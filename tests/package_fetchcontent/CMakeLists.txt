cmake_minimum_required(VERSION 3.21)
project(mwtl_fetchcontent_consumer LANGUAGES CXX)
include(FetchContent)
set(MWTL_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(MWTL_BUILD_TESTS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(mwtl SOURCE_DIR "${MWTL_SOURCE_DIR}")
FetchContent_MakeAvailable(mwtl)
add_executable(mwtl_fetchcontent_consumer ../package_consumer/main.cpp)
target_compile_features(mwtl_fetchcontent_consumer PRIVATE cxx_std_20)
target_link_libraries(mwtl_fetchcontent_consumer PRIVATE mwtl::mwtl)
add_executable(mwtl_fetchcontent_printing_consumer ../package_consumer/printing_main.cpp)
target_compile_features(mwtl_fetchcontent_printing_consumer PRIVATE cxx_std_20)
target_link_libraries(mwtl_fetchcontent_printing_consumer PRIVATE mwtl::printing)
